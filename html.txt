// QUOTA TRACKER

const QUOTA_LIMITS = { text: 20, video: 1 };
const WINDOW_MS = 60000;

const quotaTracker = {
    text: [], video: [],
    record(type) {
        this[type].push(Date.now());
        this.cleanup();
        this.render();
    },
    cleanup() {
        const cutoff = Date.now() - WINDOW_MS;
        for (const t of ['text','video']) this[t] = this[t].filter(x => x > cutoff);
    },
    canProceed(type) {
        this.cleanup();
        return this[type].length < QUOTA_LIMITS[type];
    },
    waitTime(type) {
        this.cleanup();
        if (this[type].length < QUOTA_LIMITS[type]) return 0;
        return Math.max(0, Math.ceil((this[type][0] + WINDOW_MS - Date.now()) / 1000));
    },
    render() {
        const renderItem = (type) => {
            const el = document.getElementById(`quota-${type}`);
            if (!el) return;
            const count = this[type].length;
            const limit = QUOTA_LIMITS[type];
            el.textContent = `${count}/${limit}`;
            el.className = 'value ' + (count >= limit ? 'danger' : count >= limit - 1 ? 'warn' : 'ok');
        };
        renderItem('text'); renderItem('video');
    }
};
setInterval(() => quotaTracker.render(), 1000);

function esc(s) {
    return String(s ?? "").replace(/[&<>"']/g, c => (
        {"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c]
    ));
}

async function countdown(seconds, label) {
    for (let s = seconds; s > 0; s--) {
        if (!state.isRunning) throw new Error("Annulé");
        setStatus("global-status-text", `${label} (${s}s)`);
        showCooldown(s);
        await sleep(1000);
    }
    showCooldown(0);
}

async function waitForQuota(type, label) {
    while (!quotaTracker.canProceed(type)) {
        const wait = quotaTracker.waitTime(type);
        if (wait <= 0) break;
        log(`⏸ Quota ${type} saturé. Attente ${wait}s…`, "warn");
        await countdown(wait, label || `⏸ Quota ${type}`);
    }
}

// ==================================================================
// API FETCH (429 + réseau + quota)
// ==================================================================
async function apiFetch(url, options = {}, label = "API", type = null) {
    let rateAttempt = 0, netAttempt = 0;
    if (type) await waitForQuota(type, `⏸ Quota ${type}`);

    while (true) {
        let res;
        try {
            res = await fetch(url, options);
        } catch (netErr) {
            if (netAttempt >= MAX_NETWORK_RETRIES) throw new Error(`Erreur réseau : ${netErr.message}`);
            const waitSec = NETWORK_ERROR_BACKOFF_SEC[netAttempt] || 60;
            log(`⚠️ ${label} : erreur réseau. Retry ${netAttempt+1}/${MAX_NETWORK_RETRIES} dans ${waitSec}s…`, "warn");
            await countdown(waitSec, "🔌 Erreur réseau");
            netAttempt++;
            continue;
        }
        netAttempt = 0;
        if (res.status !== 429) {
            if (type && res.ok) quotaTracker.record(type);
            return res;
        }
        if (rateAttempt >= MAX_RATE_LIMIT_RETRIES) return res;
        const waitSec = RATE_LIMIT_BACKOFF_SEC[rateAttempt] || 60;
        log(`⚠️ ${label} : RATE LIMIT 429. Attente ${waitSec}s (${rateAttempt+1}/${MAX_RATE_LIMIT_RETRIES})…`, "warn");
        await countdown(waitSec, "⏸ Rate limit");
        rateAttempt++;
    }
}

function fileToDataUri(file) {
    return new Promise((resolve, reject) => {
        const reader = new FileReader();
        reader.onload = (e) => resolve(e.target.result);
        reader.onerror = () => reject(new Error("Erreur lecture fichier."));
        reader.readAsDataURL(file);
    });
}

function toggleStyle(id, checked) {
    if (checked) state.selectedStyles.add(id);
    else state.selectedStyles.delete(id);
    document.querySelectorAll(".style-checkbox").forEach(el => {
        if (el.dataset.id === id) el.classList.toggle("checked", checked);
    });
    updateGenerateBtn();
}

function updateGenerateBtn() {

        : `🎬 Générer ${state.selectedStyles.size} vidéo${state.selectedStyles.size > 1 ? 's' : ''}`;

async function handleFile(file) {
    if (!file.type.startsWith("image/")) { alert("Fichier non image."); return; }
    if (file.size > 10 * 1024 * 1024) { alert("Image trop lourde (max 10 MB)."); return; }
    try {
        const dataUri = await fileToDataUri(file);
        state.imageDataUri = dataUri;
        previewThumb.src = dataUri;
        previewThumb.classList.add("visible");
        log(`Image chargée (${(file.size / 1024).toFixed(0)} Ko).`, "ok");
        updateGenerateBtn();
    } catch (err) { alert("Erreur : " + err.message); }
}

function renderQueue() {
    const panel = document.getElementById("queue-panel");
    if (state.queue.length === 0) { panel.classList.add("hidden"); return; }
    panel.classList.remove("hidden");

    panel.innerHTML = state.queue.map((item, i) => {
        let icon = "⏳", statusText = "En attente";
        if (item.status === "processing") { icon = "◐"; statusText = item.progress || "Génération…"; }
        else if (item.status === "done") { icon = "✅"; statusText = "Terminé"; }
        else if (item.status === "failed") { icon = "❌"; statusText = "Échec"; }

        const cls = item.status === "pending" ? "pending"
                  : item.status === "processing" ? "processing"
                  : item.status === "done" ? "done" : "failed";

        const durationSec = (item.numFrames / 24).toFixed(1);
        return `
            <div class="queue-item ${cls}">
                <div class="queue-icon">${icon}</div>
                <div class="queue-info">
                    <div class="queue-style">${item.style.icon} ${esc(item.style.name)}</div>
                    <div class="queue-meta">
                        <span>⏱ ${durationSec}s</span>
                        <span>🎬 ${esc(item.motionKey)}</span>
                    </div>
                </div>
                <div class="queue-status">${statusText}</div>
            </div>
        `;
    }).join("");
}

async function startGeneration() {
    if (state.isRunning) return;

    if (!state.imageDataUri) { alert("Choisissez une image."); return; }
    if (state.selectedStyles.size === 0) { alert("Cochez au moins un style."); return; }

    state.isRunning = true;
    state.completed = 0;

    const durationSelect = document.getElementById("duration-select");
    const motionSelect = document.getElementById("motion-select");

    state.durationFrames = parseInt(durationSelect.value, 10) || 153;
    state.motionKey = motionSelect.value;

    // Construire la file d'attente
    const selectedStyles = Array.from(state.selectedStyles)
        .map(id => STYLES.find(s => s.id === id))
        .filter(Boolean);

    state.queue = selectedStyles.map(style => ({
        style,
        status: "pending",
        videoUrl: null,
        videoId: null,
        progress: null,
        numFrames: state.durationFrames,
        motionKey: state.motionKey,
        prompt: null
    }));

    // Nettoyer l'interface
    document.getElementById("video-grid").innerHTML = "";
    document.getElementById("gallery-panel").classList.add("hidden");
    document.getElementById("log-box").innerHTML = "";
    document.getElementById("status-panel").classList.remove("hidden");
    document.getElementById("queue-panel").classList.remove("hidden");

    renderQueue();
    updateGenerateBtn();

    const durationSec = (state.durationFrames / 24).toFixed(1);
    log(`🎬 Démarrage — ${state.queue.length} vidéos à générer`, "hl");
    log(`Durée par vidéo : ${durationSec}s | Mouvement : ${state.motionKey}`, "");
    log(`Estimation totale : ~${Math.ceil(state.queue.length * (parseInt(durationSec) + 60) / 60)} minutes`, "");

    try {
        // Pause pré-vidéo (respect RPM vidéo = 1/min)
        if (quotaTracker.video.length > 0) {
            const wait = quotaTracker.waitTime("video");
            if (wait > 0) {
                log(`Pause de ${wait}s avant le premier appel vidéo (quota vidéo saturé)…`, "warn");
                await countdown(wait, "⏸ Quota vidéo");
            }
        } else {
            log(`Pause de ${PRE_VIDEO_DELAY_MS / 1000}s avant le premier clip…`, "warn");
            await countdown(Math.floor(PRE_VIDEO_DELAY_MS / 1000), "⏸ Pause pré-vidéo");
        }

        // Génération séquentielle
        for (let i = 0; i < state.queue.length; i++) {
            if (!state.isRunning) break;
            const item = state.queue[i];

            log(`━━━ Vidéo ${i+1}/${state.queue.length} — ${item.style.icon} ${item.style.name} ━━━`, "hl");
            item.status = "processing";
            item.progress = "Création…";
            renderQueue();

            // Construire le prompt
            const finalPrompt = buildPrompt(item.style, state.motionKey);
            item.prompt = finalPrompt;

            // Créer la tâche vidéo
            let videoId;
            try {
                const body = {
                    model: MODELS.video,
                    prompt: finalPrompt,
                    image: state.imageDataUri,
                    num_frames: item.numFrames,
                    frame_rate: FRAME_RATE
                };

                const res = await apiFetch(`${BASE_URL}/videos`, {
                    method: "POST", headers: apiHeaders(), body: JSON.stringify(body)
                }, `Vidéo ${i+1} · Création`, "video");

                if (!res.ok) {
                    const errText = await res.text();
                    throw new Error(`HTTP ${res.status} — ${errText.slice(0, 120)}`);
                }
                const data = await res.json();
                videoId = data.video_id || data.id || data.task_id;
                if (!videoId) throw new Error("Pas de video_id.");
                item.videoId = videoId;
                log(`Vidéo ${i+1} : tâche créée (${String(videoId).slice(0, 25)}…)`, "ok");
            } catch (err) {
                log(`Vidéo ${i+1} : échec création — ${err.message}`, "err");
                item.status = "failed";
                item.progress = "Échec création";
                renderQueue();
                if (i < state.queue.length - 1 && state.isRunning) {
                    await countdown(Math.floor(INTER_VIDEO_DELAY_MS / 1000), "⏸ Pause inter-vidéo");
                }
                continue;
            }

            // Polling
            try {
                const url = await pollVideo(videoId, `Vidéo ${i+1}`, item);
                item.status = "done";
                item.videoUrl = url;
                item.progress = "Terminé";
                state.completed++;
                renderQueue();
                addVideoToGallery(item);
                document.getElementById("gallery-panel").classList.remove("hidden");
                log(`Vidéo ${i+1} : ✅ TERMINÉE`, "ok");
            } catch (err) {
                log(`Vidéo ${i+1} : ${err.message}`, "err");
                item.status = "failed";
                item.progress = "Échec";
                renderQueue();
            }

            updateGenerateBtn();

            // Pause inter-vidéo (RPM vidéo = 1/min)
            if (i < state.queue.length - 1 && state.isRunning) {
                log(`Pause de ${INTER_VIDEO_DELAY_MS / 1000}s avant la vidéo suivante…`, "warn");
                await countdown(Math.floor(INTER_VIDEO_DELAY_MS / 1000), "⏸ Pause inter-vidéo");
            }
        }

        log(`✦ Workflow terminé — ${state.completed}/${state.queue.length} vidéos générées ✦`,
            state.completed > 0 ? "ok" : "err");
        setStatus("global-status-text", `✦ ${state.completed}/${state.queue.length} vidéos générées ✦`);

    } catch (err) {
        log(`Erreur globale : ${err.message}`, "err");
        setStatus("global-status-text", "Erreur : " + err.message);
    } finally {
        state.isRunning = false;
        showCooldown(0);
        updateGenerateBtn();
    }
}

// ==================================================================
// POLLING VIDÉO
// ==================================================================
async function pollVideo(videoId, label, item) {
    // Attente initiale
    let remaining = Math.floor(FIRST_POLL_DELAY_MS / 1000);
    while (remaining > 0 && state.isRunning) {
        item.progress = `Attente… ${remaining}s`;
        renderQueue();
        await sleep(1000);
        remaining--;
    }
    if (!state.isRunning) throw new Error("Annulé");

    for (let attempt = 0; attempt < MAX_POLL_ATTEMPTS; attempt++) {
        if (!state.isRunning) throw new Error("Annulé");
        if (attempt > 0) {
            let r = Math.floor(POLL_INTERVAL_MS / 1000);
            while (r > 0 && state.isRunning) {
                item.progress = `Vérif. ${attempt+1} — ${r}s`;
                renderQueue();
                await sleep(1000);
                r--;
            }
        }

        const url = `${RAW_URL}/agnesapi?video_id=${encodeURIComponent(videoId)}&model_name=${encodeURIComponent(MODELS.video)}`;
        try {
            const r = await apiFetch(url, { method: "GET", headers: apiHeaders() }, `${label} · Poll`);
            const d = await r.json();
            const status = d.status || "unknown";
            const progress = d.progress ?? 0;
            log(`${label} : statut="${status}", progression=${progress}%`);
            item.progress = `${progress}%`;
            renderQueue();

            if (status === "completed" || status === "succeeded") {
                const videoUrl = d.metadata?.url || d.url || d.output?.url;
                if (!videoUrl) throw new Error("Terminé mais pas d'URL.");
                return videoUrl;
            }
            if (status === "failed" || status === "error")
                throw new Error("La génération a échoué.");
        } catch (err) {
            if (err.message.includes("échoué") || err.message.includes("URL")) throw err;
            log(`${label} : erreur polling, on réessaie…`, "warn");
        }
    }
    throw new Error("Délai maximal dépassé.");
}
